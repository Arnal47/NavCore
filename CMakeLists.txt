cmake_minimum_required(VERSION 3.20)
project(NavCore VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror -Wno-missing-field-initializers -pedantic)
add_library(navcore src/road_graph.cpp src/csv_loader.cpp src/router.cpp src/osm_loader.cpp src/map_matcher.cpp src/route_integration.cpp src/spatial_index.cpp src/road_event.cpp src/route_monitor.cpp)
target_include_directories(navcore PUBLIC include)
add_executable(navcore_cli src/main.cpp)
target_link_libraries(navcore_cli PRIVATE navcore)
add_executable(navcore_match src/match_main.cpp)
target_link_libraries(navcore_match PRIVATE navcore)
add_executable(navcore_integrate src/integrate_main.cpp)
target_link_libraries(navcore_integrate PRIVATE navcore)
add_executable(navcore_spatial_benchmark tools/spatial_benchmark.cpp)
target_link_libraries(navcore_spatial_benchmark PRIVATE navcore)
add_executable(navcore_benchmark tools/benchmark.cpp)
target_link_libraries(navcore_benchmark PRIVATE navcore)
add_executable(navcore_match_benchmark tools/match_benchmark.cpp)
target_link_libraries(navcore_match_benchmark PRIVATE navcore)
enable_testing()
add_executable(navcore_tests tests/test_navcore.cpp)
target_link_libraries(navcore_tests PRIVATE navcore)
add_test(NAME navcore_tests COMMAND navcore_tests)
add_executable(navcore_v2_tests tests/test_v2_map_matching.cpp)
target_link_libraries(navcore_v2_tests PRIVATE navcore)
add_test(NAME navcore_v2_tests COMMAND navcore_v2_tests)
add_test(NAME integration_smoke COMMAND navcore_integrate --osm ${CMAKE_SOURCE_DIR}/data/sample_map.osm --gps ${CMAKE_SOURCE_DIR}/data/sample_trace.csv)
add_executable(navcore_v3_tests tests/test_v3_spatial.cpp)
target_link_libraries(navcore_v3_tests PRIVATE navcore)
add_test(NAME navcore_v3_tests COMMAND navcore_v3_tests)
add_executable(navcore_reroute_tests tests/test_v3_reroute.cpp)
target_link_libraries(navcore_reroute_tests PRIVATE navcore)
add_test(NAME navcore_reroute_tests COMMAND navcore_reroute_tests)
add_test(NAME cli_smoke COMMAND navcore_cli --map ${CMAKE_SOURCE_DIR}/data/sample_map.csv --from A --to F --algo astar --cost time)
add_test(NAME osm_smoke COMMAND navcore_cli --osm ${CMAKE_SOURCE_DIR}/data/sample_map.osm --from-node 1 --to-node 3 --algo astar --cost time)
add_test(NAME match_smoke COMMAND navcore_match --osm ${CMAKE_SOURCE_DIR}/data/sample_map.osm --gps ${CMAKE_SOURCE_DIR}/data/sample_trace.csv)
set_tests_properties(navcore_tests navcore_v2_tests navcore_v3_tests navcore_reroute_tests integration_smoke cli_smoke osm_smoke match_smoke PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
